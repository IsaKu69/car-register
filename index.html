// ========================================================= // 🚗 API หลังบ้าน: ระบบลงทะเบียนผู้ใช้งาน & จัดการยานพาหนะส่วนกลาง // ========================================================= function doGet(e) { var action = e.parameter.action; var callback = e.parameter.callback; var result = {}; try { // ------------------------------------------------------- // หมวดที่ 1: ระบบจัดการผู้ใช้งาน (แผ่นงาน setup) // ------------------------------------------------------- if (action === "checkUser") { result = checkUserRegistration(e.parameter.lineUid); } else if (action === "getRoster") { result = getPreloadedRoster(); } // ------------------------------------------------------- // หมวดที่ 2: ระบบจัดการยานพาหนะ/เครื่องจักรกล (แผ่นงาน vehicles) // ------------------------------------------------------- // 2.1 ดึงข้อมูลรถรายคันจากการสแกน QR Code หรือระบุรหัส/ทะเบียน else if (action === "getVehicle") { result = getVehicleDetail(e.parameter.carId || e.parameter.plateNumber); } // 2.2 ดึงรายการรถทั้งหมดที่สถานะ "พร้อมใช้งาน" (แสดงใน Dropdown) else if (action === "getAvailableVehicles") { result = getAvailableVehicles(); } // 2.3 ดึงข้อมูลประวัติรถทั้งหมด (สำหรับหน้าแอดมิน/จัดการรายการรถ) else if (action === "getAllVehicles") { result = getAllVehicles(); } else { result = { status: "ready", timestamp: new Date().toISOString() }; } } catch (err) { result = { success: false, error: err.toString() }; } // ส่งกลับในรูปแบบ JSONP สำหรับฝั่ง Frontend (GitHub Pages) เพื่อเลี่ยงปัญหา CORS if (callback) { return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var payload = JSON.parse(e.postData.contents); var action = payload.action || "registerUser"; var res = {}; // แยกการทำงานตาม action if (action === "saveVehicle") { // เพิ่มหรือแก้ไขข้อมูลประวัติรถ res = saveVehicleData(payload); } else if (action === "checkoutVehicle") { // บันทึกเริ่มนำรถออกปฏิบัติงาน (Check-out) res = processCheckoutVehicle(payload); } else { // ลงทะเบียนผู้ใช้งาน (ค่าเริ่มต้นเดิม) res = processRegistration(payload); } return ContentService.createTextOutput(JSON.stringify(res)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ========================================================= // 👤 ฟังก์ชันหมวดผู้ใช้งาน (แผ่นงาน setup) // ========================================================= function checkUserRegistration(lineUid) { try { if (!lineUid || lineUid.trim() === "") return { registered: false, error: "ไม่พบรหัส LINE UID" }; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("setup"); if (!sheet) return { registered: false, error: "ไม่พบแผ่นงาน setup" }; var lastRow = sheet.getLastRow(); if (lastRow <= 1) return { registered: false }; var data = sheet.getRange(2, 1, lastRow - 1, 8).getValues(); for (var i = 0; i < data.length; i++) { var currentUid = data[i][1] ? data[i][1].toString().trim() : ""; if (currentUid === lineUid.trim()) { return { registered: true, userData: { userId: data[i][0], fullname: data[i][2], position: data[i][3], department: data[i][4], phone: data[i][5], role: data[i][6], status: data[i][7] } }; } } return { registered: false }; } catch (err) { return { registered: false, error: err.toString() }; } } function getPreloadedRoster() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("setup"); if (!sheet) return []; var lastRow = sheet.getLastRow(); if (lastRow <= 1) return []; var data = sheet.getRange(2, 1, lastRow - 1, 7).getValues(); var roster = []; for (var i = 0; i < data.length; i++) { var uid = data[i][1] ? data[i][1].toString().trim() : ""; var name = data[i][2] ? data[i][2].toString().trim() : ""; if (name !== "" && uid === "") { roster.push({ rowNum: i + 2, userId: data[i][0] || "", fullname: name, position: data[i][3] || "", department: data[i][4] || "", phone: data[i][5] || "", role: data[i][6] || "ผู้ใช้รถทั่วไป" }); } } return roster; } catch (err) { return []; } } function processRegistration(payload) { var lock = LockService.getScriptLock(); try { lock.waitLock(15000); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("setup"); if (!sheet) return { success: false, error: "ไม่พบแผ่นงาน setup" }; var lineUid = payload.lineUid ? payload.lineUid.toString().trim() : ""; if (lineUid === "") return { success: false, error: "ไม่พบ LINE User ID" }; var lastRow = sheet.getLastRow(); if (lastRow > 1) { var existingUids = sheet.getRange(2, 2, lastRow - 1, 1).getValues(); for (var k = 0; k < existingUids.length; k++) { if (existingUids[k][0].toString().trim() === lineUid) { return { success: false, error: "บัญชี LINE นี้ถูกใช้งานลงทะเบียนไปแล้ว" }; } } } var now = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss"); if (payload.mode === "existing" && payload.rowNum) { var targetRow = parseInt(payload.rowNum); var checkUid = sheet.getRange(targetRow, 2).getValue().toString().trim(); if (checkUid !== "") { return { success: false, error: "ขออภัย รายชื่อนี้เพิ่งมีผู้ผูกสิทธิ์ไปแล้ว" }; } sheet.getRange(targetRow, 2).setValue(lineUid); if (payload.phone) sheet.getRange(targetRow, 6).setValue(payload.phone); sheet.getRange(targetRow, 8).setValue("พร้อมใช้งาน"); sheet.getRange(targetRow, 9).setValue(now); return { success: true, mode: "linked" }; } else if (payload.mode === "new") { var newUserId = payload.empId || ("USR-" + Utilities.formatString("%03d", lastRow)); var newRow = [ newUserId, lineUid, payload.fullname.trim(), payload.position.trim(), payload.department.trim(), payload.phone.trim(), payload.role || "ผู้ใช้รถทั่วไป", "พร้อมใช้งาน", now ]; sheet.appendRow(newRow); return { success: true, mode: "created" }; } return { success: false, error: "รูปแบบคำขอไม่ถูกต้อง" }; } catch (err) { return { success: false, error: err.toString() }; } finally { lock.releaseLock(); } } // ========================================================= // 🚜 ฟังก์ชันหมวดยานพาหนะ/เครื่องจักรกล (แผ่นงาน vehicles) // ========================================================= /** * ค้นหารายละเอียดรถรายคันตาม car_id หรือเลขทะเบียน (ปรับปรุงให้ค้นหาแบบยืดหยุ่น) */ function getVehicleDetail(queryKey) { try { if (!queryKey || queryKey.toString().trim() === "") { return { success: false, error: "ไม่ได้ระบุรหัสรถหรือเลขทะเบียน" }; } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("vehicles"); if (!sheet) return { success: false, error: "ไม่พบแผ่นงาน vehicles (กรุณาตรวจชื่อแท็บชีต)" }; var lastRow = sheet.getLastRow(); if (lastRow <= 1) return { success: false, error: "ยังไม่มีข้อมูลรถในระบบ" }; var data = sheet.getRange(2, 1, lastRow - 1, 11).getValues(); var rawTarget = queryKey.toString().trim().toLowerCase(); var numericTarget = rawTarget.replace(/[^0-9]/g, ""); for (var i = 0; i < data.length; i++) { var currentCarId = data[i][0] ? data[i][0].toString().trim().toLowerCase() : ""; var currentPlate = data[i][1] ? data[i][1].toString().trim().toLowerCase() : ""; var plateNumbersOnly = currentPlate.replace(/[^0-9]/g, ""); var isMatch = false; if (currentCarId === rawTarget || currentPlate === rawTarget) { isMatch = true; } else if (rawTarget.length >= 2 && (currentCarId.indexOf(rawTarget) !== -1 || currentPlate.indexOf(rawTarget) !== -1)) { isMatch = true; } else if (numericTarget.length >= 2 && plateNumbersOnly.indexOf(numericTarget) !== -1) { isMatch = true; } if (isMatch) { return { success: true, rowNum: i + 2, vehicle: { carId: data[i][0] ? data[i][0].toString().trim() : "", plateNumber: data[i][1] ? data[i][1].toString().trim() : "", type: data[i][2] || "-", brandModel: data[i][3] || "-", department: data[i][4] || "-", currentMileage: Number(data[i][5]) || 0, fuelType: data[i][6] || "-", status: data[i][7] || "พร้อมใช้งาน", driverUid: data[i][8] || "", qrCodeUrl: data[i][9] || "", note: data[i][10] || "" } }; } } return { success: false, error: "ไม่พบข้อมูลรถสำหรับคำค้น: \"" + queryKey + "\"" }; } catch (err) { return { success: false, error: err.toString() }; } } /** * ดึงรายการรถเฉพาะที่สถานะ "พร้อมใช้งาน" */ function getAvailableVehicles() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("vehicles"); if (!sheet) return { success: false, error: "ไม่พบแผ่นงาน vehicles" }; var lastRow = sheet.getLastRow(); if (lastRow <= 1) return { success: true, vehicles: [] }; var data = sheet.getRange(2, 1, lastRow - 1, 8).getValues(); var list = []; for (var i = 0; i < data.length; i++) { var status = data[i][7] ? data[i][7].toString().trim() : "พร้อมใช้งาน"; if (status === "พร้อมใช้งาน") { list.push({ rowNum: i + 2, carId: data[i][0], plateNumber: data[i][1], type: data[i][2], brandModel: data[i][3], department: data[i][4], currentMileage: Number(data[i][5]) || 0 }); } } return { success: true, vehicles: list }; } catch (err) { return { success: false, error: err.toString() }; } } /** * ดึงข้อมูลรถทั้งหมดในระบบ */ function getAllVehicles() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("vehicles"); if (!sheet) return { success: false, error: "ไม่พบแผ่นงาน vehicles" }; var lastRow = sheet.getLastRow(); if (lastRow <= 1) return { success: true, vehicles: [] }; var data = sheet.getRange(2, 1, lastRow - 1, 11).getValues(); var list = []; for (var i = 0; i < data.length; i++) { list.push({ rowNum: i + 2, carId: data[i][0], plateNumber: data[i][1], type: data[i][2], brandModel: data[i][3], department: data[i][4], currentMileage: Number(data[i][5]) || 0, fuelType: data[i][6], status: data[i][7] || "พร้อมใช้งาน", driverUid: data[i][8] || "", qrCodeUrl: data[i][9] || "", note: data[i][10] || "" }); } return { success: true, vehicles: list }; } catch (err) { return { success: false, error: err.toString() }; } } /** * บันทึกข้อมูลรถ (เพิ่มรถใหม่ หรือ แก้ไขข้อมูลรถเดิม) */ function saveVehicleData(payload) { var lock = LockService.getScriptLock(); try { lock.waitLock(15000); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("vehicles"); if (!sheet) return { success: false, error: "ไม่พบแผ่นงาน vehicles" }; var lastRow = sheet.getLastRow(); var carId = payload.carId ? payload.carId.trim() : ""; var plateNumber = payload.plateNumber ? payload.plateNumber.trim() : ""; if (!plateNumber) return { success: false, error: "กรุณาระบุเลขทะเบียนรถ" }; if (payload.rowNum && parseInt(payload.rowNum) > 1) { var row = parseInt(payload.rowNum); if (carId) sheet.getRange(row, 1).setValue(carId); sheet.getRange(row, 2).setValue(plateNumber); if (payload.type) sheet.getRange(row, 3).setValue(payload.type); if (payload.brandModel) sheet.getRange(row, 4).setValue(payload.brandModel); if (payload.department) sheet.getRange(row, 5).setValue(payload.department); if (payload.currentMileage !== undefined) sheet.getRange(row, 6).setValue(Number(payload.currentMileage)); if (payload.fuelType) sheet.getRange(row, 7).setValue(payload.fuelType); if (payload.status) sheet.getRange(row, 8).setValue(payload.status); if (payload.driverUid !== undefined) sheet.getRange(row, 9).setValue(payload.driverUid); if (payload.note !== undefined) sheet.getRange(row, 11).setValue(payload.note); return { success: true, mode: "updated" }; } else { var newCarId = carId || ("CAR-" + Utilities.formatString("%03d", lastRow)); var qrUrl = "https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=" + encodeURIComponent(newCarId); var newRow = [ newCarId, plateNumber, payload.type || "รถยนต์ทั่วไป", payload.brandModel || "-", payload.department || "-", Number(payload.currentMileage) || 0, payload.fuelType || "ดีเซล", payload.status || "พร้อมใช้งาน", payload.driverUid || "", qrUrl, payload.note || "" ]; sheet.appendRow(newRow); return { success: true, mode: "created", carId: newCarId }; } } catch (err) { return { success: false, error: err.toString() }; } finally { lock.releaseLock(); } } // ========================================================= // 📋 ฟังก์ชันหมวดบันทึกการใช้งาน (แผ่นงาน usage_logs) // ========================================================= /** * บันทึกเริ่มนำรถออกปฏิบัติงาน (Check-out) * 1. บันทึกประวัติลงแผ่นงาน usage_logs * 2. อัปเดตสถานะรถใน vehicles เป็น "กำลังใช้งาน" */ function processCheckoutVehicle(payload) { var lock = LockService.getScriptLock(); try { lock.waitLock(15000); var ss = SpreadsheetApp.getActiveSpreadsheet(); var logSheet = ss.getSheetByName("usage_logs"); var carSheet = ss.getSheetByName("vehicles"); if (!logSheet) return { success: false, error: "ไม่พบแผ่นงาน usage_logs (กรุณาสร้างแท็บชีต usage_logs)" }; if (!carSheet) return { success: false, error: "ไม่พบแผ่นงาน vehicles" }; var carId = (payload.carId || "").toString().trim(); var plateNumber = (payload.plateNumber || "").toString().trim(); var startMileage = Number(payload.startMileage); if (!carId && !plateNumber) return { success: false, error: "ไม่พบข้อมูลระบุตัวรถ" }; if (isNaN(startMileage) || startMileage < 0) return { success: false, error: "กรุณาระบุเลขไมล์เริ่มต้นให้ถูกต้อง" }; // ค้นหารถในแผ่นงาน vehicles var carLastRow = carSheet.getLastRow(); var targetRow = -1; var currentMileage = 0; if (carLastRow > 1) { var carData = carSheet.getRange(2, 1, carLastRow - 1, 8).getValues(); for (var i = 0; i < carData.length; i++) { var cId = (carData[i][0] || "").toString().trim().toLowerCase(); var pNum = (carData[i][1] || "").toString().trim().toLowerCase(); if (cId === carId.toLowerCase() || pNum === plateNumber.toLowerCase()) { targetRow = i + 2; currentMileage = Number(carData[i][5]) || 0; break; } } } if (targetRow === -1) { return { success: false, error: "ไม่พบรถคันนี้ในฐานข้อมูล" }; } // ตรวจสอบความถูกต้องของเลขไมล์เริ่มต้น if (startMileage < currentMileage) { return { success: false, error: "เลขไมล์เริ่มต้น (" + startMileage + ") ต้องไม่น้อยกว่าเลขไมล์ล่าสุด (" + currentMileage + ")" }; } // สร้าง Log ID อัตโนมัติ (LOG-YYYYMMDD-XXX) var now = new Date(); var datePrefix = Utilities.formatDate(now, "GMT+7", "yyyyMMdd"); var timeFormatted = Utilities.formatDate(now, "GMT+7", "dd/MM/yyyy HH:mm:ss"); var logLastRow = logSheet.getLastRow(); var logId = "LOG-" + datePrefix + "-" + Utilities.formatString("%03d", logLastRow); // 1. เพิ่มแถวใหม่ใน usage_logs var newLog = [ logId, carId, plateNumber, (payload.driverName || "-").toString().trim(), (payload.driverUid || "").toString().trim(), (payload.purpose || "-").toString().trim(), timeFormatted, startMileage, "", // end_time (รอ Check-in) "", // end_mileage (รอ Check-in) "", // total_distance (รอ Check-in) "", // issue_reported (รอ Check-in) "กำลังใช้งาน" ]; logSheet.appendRow(newLog); // 2. อัปเดตข้อมูลรถในแผ่นงาน vehicles carSheet.getRange(targetRow, 6).setValue(startMileage); // Col F: current_mileage carSheet.getRange(targetRow, 8).setValue("กำลังใช้งาน"); // Col H: status if (payload.driverUid) { carSheet.getRange(targetRow, 9).setValue(payload.driverUid); // Col I: driver_uid } return { success: true, logId: logId, plateNumber: plateNumber, startMileage: startMileage }; } catch (err) { return { success: false, error: err.toString() }; } finally { lock.releaseLock(); } }
